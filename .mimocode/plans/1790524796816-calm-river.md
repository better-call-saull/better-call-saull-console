# 技术债务修复计划

## 修复清单

### Fix 1: app.js:205 — 硬编码年度节点索引

**问题**：`DB.annual[3]` 假设第 4 个元素是最近节点，数组重排即失效

**修复方案**：动态查找最近截止日的节点

```javascript
// 修复前
esc(DB.annual[3].item) + '（' + DB.annual[3].deadline + '）'

// 修复后
(function() {
  var nearest = DB.annual.reduce(function(a, b) {
    return a.deadline < b.deadline ? a : b;
  });
  return esc(nearest.item) + '（' + nearest.deadline + '）';
})()
```

### Fix 6: app.js:743-745 — 清单 KPI 静态字符串

**问题**：3 个 KPI 为硬编码文本

**修复方案**：从 DB.checklist 动态查找未完成的高优先项

```javascript
// 修复前
'<div class="v" style="font-size:15px;line-height:1.5;padding-top:7px">数据分类分级</div><div class="w">网络数据安全管理条例</div>'

// 修复后（动态查找第一个未完成项）
(function() {
  var first = null;
  DB.checklist.some(function(g) {
    return g.items.some(function(it, i) {
      if (!it.done && !cklState[g.domain + "|" + i]) {
        first = { q: it.q.split('（')[0], ref: it.ref };
        return true;
      }
      return false;
    });
  });
  return first ? 
    '<div class="v" style="font-size:15px;line-height:1.5;padding-top:7px">' + esc(first.q) + '</div><div class="w">' + esc(first.ref) + '</div>' :
    '<div class="v" style="font-size:15px;line-height:1.5;padding-top:7px">全部完成</div><div class="w">—</div>';
})()
```
